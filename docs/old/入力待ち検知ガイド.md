## 基本概念

### 入力待ちとは

PyPrologプログラムが以下の述語を実行する際に発生します：

- `read_line/1`: 行単位の入力読み取り
- `get_char/1`: 1文字の入力読み取り
- `peek_char/1`: 先読み入力

### 検知のタイミング

入力待ちは以下のタイミングで検知できます：

1. **IOStreamメソッド呼び出し時**: `read_line()`や`read_char()`が呼ばれた瞬間
2. **カスタムストリーム実装**: 入力要求をインターセプト
3. **コールバック関数**: 入力が必要になった時点で外部に通知

```python
        Args:

        """

        self.input_history = []
    
    def handle_input_request(self, event: InputEvent) -> Optional[str]:
        """
        入力要求を処理
        
        Args:
            event: 入力イベント（入力タイプ、述語名等を含む）
            
        Returns:
            Optional[str]: 入力値（None = EOF）
        """

        print(f"   述語: {event.predicate_name}")
        print(f"   プロンプト: {event.prompt}")
        
        # 入力タイプに応じた処理
        if event.input_type == "char":
            result = self._get_char_input(event)
        elif event.input_type == "line": 
            result = self._get_line_input(event)

                'type': event.input_type,
                'value': result,
                'predicate': event.predicate_name
            })
        
        return result

        if self.input_source and hasattr(self.input_source, 'get_char'):
            return self.input_source.get_char()
```

```python
from pyprolog.runtime.unified_input_system import ThreadingController

class AdvancedInputHandler(InputHandler):
    """高度な入力ハンドラ（真の継続実行対応）"""
    
    def __init__(self, gui_interface=None):
        self.gui_interface = gui_interface
        self.suspended_continuations = {}
        
    def handle_input_request(self, event: InputEvent) -> Optional[str]:
        """
        真の継続実行による入力処理
        
        この時点で、Prologの実行スタックは完全に保持され、
        入力取得後に正確に同じポイントから実行再開される
        """
        continuation_id = event.request_id  # システム生成ID
        
        if self.gui_interface:
            # GUI環境での非同期入力要求
            self.gui_interface.request_input_async(
                input_type=event.input_type,
                prompt=event.prompt,
                callback=lambda value: self._resume_continuation(continuation_id, value)
            )
            
            # スレッド制御: この時点で実行は中断されるが、
            # スタックフレームは完全保持される
            return None  # 継続は後で再開される
        else:
            # 同期実行の場合
            return input(f"⚡ [継続実行] {event.prompt}")
    
    def _resume_continuation(self, continuation_id: str, input_value: str):
        """継続実行の再開"""
        # 統一入力システムに入力値を送信して実行再開
        threading_controller = ThreadingController.get_instance()
        threading_controller.resume_continuation(continuation_id, input_value)
```

```python
from pyprolog.runtime.interpreter import Runtime
from pyprolog.runtime.io_streams import IOStream

class InputDetectionStream(IOStream):
    """入力待ちを検知し、外部に通知するストリーム"""
    
    def __init__(self, input_callback=None, output_callback=None):
        self.input_callback = input_callback
        self.output_callback = output_callback
        
    def read_line(self):
        """行入力待ちを検知"""
        if self.input_callback:
            result = self.input_callback("line_input_needed")
            if result:
                return result
        
        # デフォルト: 標準入力
        return input("入力してください: ")
    
    def read_char(self):
        """文字入力待ちを検知"""
        if self.input_callback:
            result = self.input_callback("char_input_needed")
            if result:
                return result[0]
        
        return input("文字を入力: ")[0]
    
    def write_char(self, char):
        if self.output_callback:
            self.output_callback(char)
        else:
            print(char, end='')
    
    def write_term(self, term):
        text = str(term)
        if self.output_callback:
            self.output_callback(text)
        else:
            print(text, end='')
    
    # その他の必須メソッド
    def read_term(self): pass
    def peek_char(self): return 'x'
    def at_end_of_stream(self): return False

# 使用例
def input_handler(input_type):
    print(f"[検知] 入力待ち発生: {input_type}")
    
    # 様々な入力ソースから値を取得
    if input_type == "line_input_needed":
        return get_line_from_gui()  # GUI入力
    elif input_type == "char_input_needed":
        return get_char_from_network()  # ネットワーク入力
    
    return "default_value"

runtime = Runtime()
stream = InputDetectionStream(input_callback=input_handler)
runtime.io_manager.set_input_stream(stream)
runtime.io_manager.set_output_stream(stream)
```

```python
import asyncio
import concurrent.futures
from pyprolog.runtime.interpreter import Runtime

class AsyncPrologExecutor:
    """非同期Prolog実行・入力待ち検知"""
    
    def __init__(self):
        self.runtime = None
        self.input_event = asyncio.Event()
        self.input_queue = asyncio.Queue()
        self.current_input_type = None
        
    async def setup_async_runtime(self):
        """非同期対応ランタイムセットアップ"""
        self.runtime = Runtime()
        
        stream = AsyncInputDetectionStream(self)
        self.runtime.io_manager.set_input_stream(stream)
        self.runtime.io_manager.set_output_stream(stream)
    
    async def execute_query_async(self, query, program_file=None):
        """非同期クエリ実行"""
        if program_file:
            self.runtime.consult(program_file)
        
        # 別スレッドでProlog実行
        loop = asyncio.get_event_loop()
        with concurrent.futures.ThreadPoolExecutor() as executor:
            future = executor.submit(self.runtime.query, query)
            
            # 入力待ち監視
            while not future.done():
                try:
                    # 短時間で入力イベント確認
                    await asyncio.wait_for(self.input_event.wait(), timeout=0.01)
                    
                    print(f"[非同期] 入力待ち検知: {self.current_input_type}")
                    
                    # 非同期で入力処理
                    input_value = await self._handle_input_async()
                    await self.input_queue.put(input_value)
                    
                    self.input_event.clear()
                    
                except asyncio.TimeoutError:
                    continue  # タイムアウト = まだ実行中
            
            return future.result()
    
    async def _handle_input_async(self):
        """非同期入力処理"""
        # WebSocket、HTTP API、非同期GUI等からの入力
        if self.current_input_type == "line_input_needed":
            return await self._get_async_line_input()
        elif self.current_input_type == "char_input_needed":
            return await self._get_async_char_input()
        
        return "async_default"
    
    async def _get_async_line_input(self):
        """非同期行入力"""
        # 実際の実装例:
        # - WebSocketからの入力待ち
        # - HTTP APIポーリング
        # - 非同期ファイル読み込み
        await asyncio.sleep(0.1)  # 模擬遅延
        return "async_line_input"
    
    async def _get_async_char_input(self):
        """非同期文字入力"""
        await asyncio.sleep(0.1)
        return "a"

class AsyncInputDetectionStream(IOStream):
    """非同期入力検知ストリーム"""
    
    def __init__(self, executor):
        self.executor = executor
        
    def read_line(self):
        """非同期入力待ち通知（同期メソッド内）"""
        self.executor.current_input_type = "line_input_needed"
        
        # イベント設定（非同期側で検知）
        asyncio.run_coroutine_threadsafe(
            self._set_input_event(), 
            asyncio.get_event_loop()
        )
        
        # 入力結果を同期的に取得
        try:
            return asyncio.run_coroutine_threadsafe(
                self.executor.input_queue.get(),
                asyncio.get_event_loop()
            ).result(timeout=10.0)
        except concurrent.futures.TimeoutError:
            return "timeout_input"
    
    async def _set_input_event(self):
        """入力イベント設定"""
        self.executor.input_event.set()
    
    def read_char(self):
        """非同期文字入力待ち"""
        self.executor.current_input_type = "char_input_needed"
        asyncio.run_coroutine_threadsafe(
            self._set_input_event(),
            asyncio.get_event_loop()
        )
        
        try:
            result = asyncio.run_coroutine_threadsafe(
                self.executor.input_queue.get(),
                asyncio.get_event_loop()
            ).result(timeout=10.0)
            return result[0] if result else 'x'
        except concurrent.futures.TimeoutError:
            return 'x'
    
    # その他の必須メソッド
    def write_char(self, char): print(char, end='')
    def write_term(self, term): print(str(term), end='')
    def read_term(self): pass
    def peek_char(self): return 'x'
    def at_end_of_stream(self): return False
```

```python
#!/usr/bin/env python3
"""
PyProlog入力待ち検知の基本例
"""

from pyprolog.runtime.interpreter import Runtime
from pyprolog.runtime.io_streams import IOStream

class BasicInputDetector(IOStream):
    """基本的な入力待ち検知"""
    
    def __init__(self):
        self.input_history = []
        self.output_history = []
    
    def read_line(self):
        print("📥 [検知] 行入力が要求されました")
        
        # ここで外部システムから入力を取得
        value = self._get_input_from_source()
        
        # 履歴記録
        self.input_history.append(value)
        print(f"✅ [提供] 入力値: {value}")
        
        return value
    
    def _get_input_from_source(self):
        """入力ソース（カスタマイズ可能）"""
        # 例1: GUI入力
        # return tkinter.simpledialog.askstring("入力", "値を入力:")
        
        # 例2: ファイルから自動入力
        # return next(file_reader, "EOF")
        
        # 例3: ネットワークから入力
        # return http_client.get_input()
        
        # デモ用: 標準入力
        return input("🎯 入力してください: ")
    
    def read_char(self):
        print("📥 [検知] 文字入力が要求されました")
        value = input("🎯 文字を入力: ")
        return value[0] if value else 'x'
    
    def write_char(self, char):
        self.output_history.append(char)
        print(char, end='')
    
    def write_term(self, term):
        text = str(term)
        self.output_history.append(text)
        print(text, end='')
    
    # その他必須メソッド
    def read_term(self): pass
    def peek_char(self): return 'x'
    def at_end_of_stream(self): return False

def demo_basic_detection():
    """基本検知デモ"""
    runtime = Runtime()
    detector = BasicInputDetector()
    
    # ストリーム設定
    runtime.io_manager.set_input_stream(detector)
    runtime.io_manager.set_output_stream(detector)

    greet_user :-
        write('お名前を入力してください: '),
        read_line(Name),
        write('こんにちは、'),
        write(Name),
        write('さん！'), nl.

    print("=== 基本的な入力待ち検知デモ ===")

    print(f"\n📊 実行結果: {'成功' if results else '失敗'}")
    print(f"📝 入力履歴: {detector.input_history}")
    print(f"📤 出力履歴: {''.join(detector.output_history)}")

if __name__ == "__main__":
    demo_basic_detection()
```

```python
import queue

class WebInputDetector(IOStream):
    """Web API統合入力検知"""
    
    def __init__(self):
        self.input_queue = queue.Queue()
        self.waiting_for_input = False
        
    def read_line(self):
        """Web APIからの入力待ち"""
        self.waiting_for_input = True
        
        # Web経由での入力待ち（polling方式）
        try:
            value = self.input_queue.get(timeout=30.0)
            self.waiting_for_input = False
            return value
        except queue.Empty:
            self.waiting_for_input = False
            return "timeout"
    
    def provide_web_input(self, value):
        """Web APIから呼び出される入力提供"""
        if self.waiting_for_input:
            self.input_queue.put(value)
            return {"status": "accepted"}
        return {"status": "not_waiting"}

# Flask統合例
app = Flask(__name__)
web_detector = WebInputDetector()

@app.route('/provide_input', methods=['POST'])
def provide_input():
    data = request.json
    result = web_detector.provide_web_input(data.get('value', ''))
    return jsonify(result)
```

```python
from pyprolog.runtime.io_streams import IOStream

class ComprehensiveTestDetector(IOStream):
    """総合テスト用入力検知"""
    
    def __init__(self):
        self.events = []  # イベントログ
        self.test_inputs = []
        self.input_index = 0
        
    def add_test_input(self, value):
        """テスト入力を追加"""
        self.test_inputs.append(str(value))
    
    def read_line(self):
        """入力待ち検知とログ記録"""
        timestamp = time.time()
        self.events.append({
            'type': 'input_request',
            'method': 'read_line',
            'timestamp': timestamp
        })
        
        print(f"📥 [{timestamp:.2f}] 行入力要求を検知")
        
        if self.input_index < len(self.test_inputs):
            value = self.test_inputs[self.input_index]
            self.input_index += 1
            
            self.events.append({
                'type': 'input_provided',
                'value': value,
                'timestamp': time.time()
            })
            
            print(f"✅ 入力提供: {value}")
            return value
        
        print("⚠️ 入力不足")
        return "default"
    
    def read_char(self):
        """文字入力検知"""
        timestamp = time.time()
        self.events.append({
            'type': 'input_request',
            'method': 'read_char',
            'timestamp': timestamp
        })
        
        print(f"📥 [{timestamp:.2f}] 文字入力要求を検知")
        
        if self.input_index < len(self.test_inputs):
            value = self.test_inputs[self.input_index]
            self.input_index += 1
            return value[0] if value else 'x'
        
        return 'x'
    
    def write_char(self, char):
        """出力ログ記録"""
        self.events.append({
            'type': 'output',
            'method': 'write_char',
            'value': char,
            'timestamp': time.time()
        })
        print(char, end='')
    
    def write_term(self, term):
        """項出力ログ記録"""
        text = str(term)
        self.events.append({
            'type': 'output',
            'method': 'write_term',
            'value': text,
            'timestamp': time.time()
        })
        print(text, end='')
    
    def get_event_summary(self):
        """イベント要約"""
        input_events = [e for e in self.events if e['type'] == 'input_request']
        output_events = [e for e in self.events if e['type'] == 'output']
        
        return {
            'total_events': len(self.events),
            'input_requests': len(input_events),
            'output_events': len(output_events),
            'timeline': self.events
        }
    
    # その他必須メソッド
    def read_term(self): pass
    def peek_char(self): return 'x'  
    def at_end_of_stream(self): return False
```

**Before (従来のカスタムストリーム):**
```python
# 旧方式
class OldInputDetector(IOStream):
    def read_line(self):
        return custom_input_source.get_line()

```

**After (統一入力システム):**
```python
# 新方式
class NewInputHandler(InputHandler):
    def handle_input_request(self, event: InputEvent) -> Optional[str]:
        if event.input_type == "line":
            return custom_input_source.get_line()
        return None

runtime.io_manager.set_input_handler(NewInputHandler())
```
